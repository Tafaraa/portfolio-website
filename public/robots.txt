# One group, applying to every crawler. Named-agent groups were removed on
# purpose: a crawler obeys only the most specific group that matches it, so the
# old "User-agent: Googlebot / Allow: /" block overrode the Disallow rules below
# and invited Google straight into /admin and /api/. A single wildcard group
# cannot drift that way.
User-agent: *
Allow: /
Disallow: /private/
Disallow: /admin
Disallow: /api/
